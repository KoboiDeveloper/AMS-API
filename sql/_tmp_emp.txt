


/* =========================================================
   EMPLOYEE
   ========================================================= */
CREATE PROCEDURE dbo.xsp_ams_new_rbac_employee_insert
(
      @p_nik varchar(30)
    , @p_employee_name varchar(150)
    , @p_employee_type varchar(30) = NULL
    , @p_branch_id varchar(30) = NULL
    , @p_store_id varchar(30) = NULL
    , @p_job_title varchar(100) = NULL
    , @p_email varchar(150) = NULL
    , @p_phone varchar(50) = NULL
    , @p_created_by varchar(30) = NULL
    , @bitparam varchar(max) OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        SET @p_nik = UPPER(LTRIM(RTRIM(ISNULL(@p_nik,''))));
        IF @p_nik = '' OR LTRIM(RTRIM(ISNULL(@p_employee_name,''))) = ''
        BEGIN
            SET @bitparam = (SELECT '4002500' responseCode, 'nik and employee_name are required' responseMessage FOR JSON PATH, WITHOUT_ARRAY_WRAPPER);
            RETURN;
        END
        IF EXISTS (SELECT 1 FROM dbo.RBAC_Employee WHERE nik = @p_nik)
        BEGIN
            SET @bitparam = (SELECT '4002500' responseCode, 'nik already exists' responseMessage FOR JSON PATH, WITHOUT_ARRAY_WRAPPER);
            RETURN;
        END
        INSERT INTO dbo.RBAC_Employee(nik, employee_name, employee_type, branch_id, store_id, job_title, email, phone, created_by)
        VALUES(@p_nik, @p_employee_name, NULLIF(@p_employee_type,''), NULLIF(@p_branch_id,''), NULLIF(@p_store_id,''), NULLIF(@p_job_title,''), NULLIF(@p_email,''), NULLIF(@p_phone,''), NULLIF(@p_created_by,''));
        SET @bitparam = (SELECT '2002500' responseCode, 'success' responseMessage, JSON_QUERY((SELECT @p_nik nik FOR JSON PATH, WITHOUT_ARRAY_WRAPPER)) data FOR JSON PATH, WITHOUT_ARRAY_WRAPPER);
    END TRY
    BEGIN CATCH
        SET @bitparam = (SELECT '4002500' responseCode, ERROR_MESSAGE() responseMessage FOR JSON PATH, WITHOUT_ARRAY_WRAPPER);
    END CATCH
END


(1 rows affected)
